let texts = [];
let answers = {};
let current = null;
let total = 0;
let completed = 0;

let step = 1;

// SAVE TEXT
function saveText() {
  let title = document.getElementById("title").value;
  let content = document.getElementById("inputText").value;

  texts.push({ title, content });
  displayTexts();
}

// DISPLAY LIBRARY
function displayTexts() {
  let list = document.getElementById("textList");
  list.innerHTML = "";

  texts.forEach((t, i) => {
    list.innerHTML += `<div class="card" onclick="loadText(${i})">${t.title}</div>`;
  });
}

// LOAD TEXT
function loadText(index) {
  let textObj = texts[index];
  total = 0;
  completed = 0;
  answers = {};
  step = 1;

  let parts = textObj.content.split(/\[(.*?)\]/);
  let display = "";

  for (let i = 0; i < parts.length; i++) {
    if (i % 2 === 1) {
      let id = "p" + total;
      display += `<span class="highlight" onclick="selectPart('${id}')">${parts[i]}</span>`;
      answers[id] = "";
      total++;
    } else {
      display += parts[i];
    }
  }

  total += 10;

  document.getElementById("textDisplay").innerHTML = display;
  document.getElementById("textTitle").innerText = textObj.title;

  document.getElementById("library").style.display = "none";
  document.getElementById("game").style.display = "block";

  updateStepUI();
}

// SELECT PART
function selectPart(id) {
  if (step !== 2) {
    alert("🔒 Termine d'abord l'étape 1 !");
    return;
  }
  current = id;
  document.getElementById("answerBox").value = answers[id];
}

// WRITE ANALYSIS
document.getElementById("answerBox").addEventListener("input", function() {
  if (current && step === 2) {
    if (answers[current] === "" && this.value !== "") completed++;
    answers[current] = this.value;
    updateProgress();
  }
});

// UPDATE PROGRESS
function updateProgress() {
  let percent = Math.round((completed / total) * 100);
  document.getElementById("bar").style.width = percent + "%";
  document.getElementById("progressText").innerText = "Progression : " + percent + "%";
}

// STEP CONTROL
function nextStep() {
  if (step === 1) {
    let problem = document.getElementById("problem").value;
    let intro = document.getElementById("intro").value;
    let plan = document.getElementById("plan").value;

    if (!problem || !intro || !plan) {
      alert("❌ Complète la problématique, l'introduction et l'annonce du plan !");
      return;
    }

    step = 2;
  }

  else if (step === 2) {
    let allDone = Object.values(answers).every(v => v !== "");

    if (!allDone) {
      alert("❌ Termine toutes les analyses !");
      return;
    }

    step = 3;
  }

  updateStepUI();
}

// UPDATE UI PER STEP
function updateStepUI() {
  document.getElementById("step1").style.display = (step === 1) ? "block" : "none";
  document.getElementById("step2").style.display = (step === 2) ? "block" : "none";
  document.getElementById("step3").style.display = (step === 3) ? "block" : "none";
}

// FINAL CHECK
function finishOral() {
  let conclusion = document.getElementById("conclusion").value;
  let ouverture = document.getElementById("ouverture").value;

  if (!conclusion || !ouverture) {
    alert("❌ Ajoute la conclusion et l'ouverture !");
    return;
  }

  alert("🔥 Parfait ! Tu es prêt(e) pour l’oral !");
}

// BACK
function goBack() {
  location.reload();
}
