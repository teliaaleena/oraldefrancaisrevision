<script>

let texts = JSON.parse(localStorage.getItem("texts")) || [];
let answers = {};
let modelAnswers = {};
let difficultParts = JSON.parse(localStorage.getItem("difficult")) || [];
let current = null;
let step = 1;

// SAVE TEXT
function saveText(){
let t = document.getElementById("title").value;
let c = document.getElementById("inputText").value;

texts.push({title:t, content:c});
localStorage.setItem("texts", JSON.stringify(texts));
displayTexts();
}

// DISPLAY
function displayTexts(){
let list="";
texts.forEach((t,i)=>{
list+=`<div class="card" onclick="loadText(${i})">${t.title}</div>`;
});
document.getElementById("textList").innerHTML=list;
}

// LOAD TEXT
function loadText(i){

let txt=texts[i];
answers={}; modelAnswers={};

let parts=txt.content.split(/\[(.*?)\]/);
let display=""; let count=0;

for(let i=0;i<parts.length;i++){
if(i%2==1){
let split=parts[i].split("|");
let q=split[0].trim();
let m=split[1]?split[1].trim():"";

let id="p"+count;

display+=`<span class="highlight" onclick="selectPart('${id}')">${q}</span>`;
answers[id]="";
modelAnswers[id]=m;
count++;
}else display+=parts[i];
}

document.getElementById("textDisplay").innerHTML=display;
document.getElementById("textTitle").innerText=txt.title;

document.getElementById("library").style.display="none";
document.getElementById("game").style.display="block";
step=1; updateStep();
}

// SELECT PART
function selectPart(id){
if(step!==2){alert("Fais d'abord l'introduction");return;}
current=id;
}

// CORRECTION
function toggleCorrection(){
if(!current)return;
alert("Correction: "+modelAnswers[current]);
}

// SELF MARK
function markCorrect(){
if(!current)return;
updateProgress();
}

function markWrong(){
if(!current)return;
difficultParts.push(current);
localStorage.setItem("difficult", JSON.stringify(difficultParts));
}

// HARD MODE
function startHardMode(){
alert("Mode parties difficiles !");
}

// RANDOM TEXT
function randomText(){
let i=Math.floor(Math.random()*texts.length);
loadText(i);
}

// STEPS
function nextStep(){
if(step==1){
let p=document.getElementById("problem").value;
let i=document.getElementById("intro").value;
let pl=document.getElementById("plan").value;

if(!p||!i||!pl){
alert("Complète l'intro !");
return;
}
step=2;
}
else if(step==2){
step=3;
}
updateStep();
}

function updateStep(){
document.getElementById("step1").style.display=step==1?"block":"none";
document.getElementById("step2").style.display=step==2?"block":"none";
document.getElementById("step3").style.display=step==3?"block":"none";
}

// PROGRESS
function updateProgress(){
let percent=Math.floor(Math.random()*100);
document.getElementById("bar").style.width=percent+"%";
document.getElementById("progressText").innerText="Progression : "+percent+"%";
}

// FINISH
function finish(){
let c=document.getElementById("conclusion").value;
let o=document.getElementById("ouverture").value;

if(!c||!o){
alert("Ajoute conclusion !");
return;
}
alert("🔥 Prêt pour l'oral !");
}

// BACK
function goBack(){
location.reload();
}

displayTexts();

</script>
