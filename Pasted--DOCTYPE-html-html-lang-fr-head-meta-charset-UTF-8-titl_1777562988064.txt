<script>

/* ===== DATA ===== */

const texts = [
  "Le Dormeur du Val",
  "Les Tahitiens",
  "Un Rêve",
  "Manon Lescaut",
  "Lettres persanes",
  "Misanthrope"
];

const corrections = {
  "Le Dormeur du Val": "Correction complète du Dormeur du Val...",
  "Les Tahitiens": "Correction complète des Tahitiens...",
  "Un Rêve": "Correction complète de Un Rêve..."
};

/* ===== FUNCTIONS ===== */

function saveText() {
  localStorage.setItem("text", document.getElementById("userText").value);
}

function editText() {
  document.getElementById("userText").disabled = false;
}

function clearText() {
  document.getElementById("userText").value = "";
  localStorage.removeItem("text");
}

function finishText() {
  let current = document.getElementById("result").innerText;
  document.getElementById("finalCorrection").innerText = corrections[current] || "Correction non disponible";
}

function spin() {
  let random = texts[Math.floor(Math.random() * texts.length)];
  document.getElementById("result").innerText = random;

  // load correction preview behind
  document.getElementById("correctionText").innerText =
    corrections[random] || "";
}

/* NAV */
function prevStep() {
  alert("Retour étape 1");
}

function nextStep() {
  alert("Aller étape 3");
}

/* LOAD SAVED */
window.onload = () => {
  let saved = localStorage.getItem("text");
  if (saved) document.getElementById("userText").value = saved;
};

</script>
