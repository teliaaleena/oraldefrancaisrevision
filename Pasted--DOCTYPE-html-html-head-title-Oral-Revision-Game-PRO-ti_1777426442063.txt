<script>
let texts = [];
let answers = {};
let current = null;
let total = 0;
let completed = 0;

// SAVE TEXT
function saveText() {
  let title = document.getElementById("title").value;
  let content = document.getElementById("inputText").value;

  texts.push({ title, content });
  displayTexts();
}

// DISPLAY LIBRARY
function displayTexts() {
  let list = document.getElementById("textList");
  list.innerHTML = "";

  texts.forEach((t, i) => {
    list.innerHTML += `<div class="card" onclick="loadText(${i})">${t.title}</div>`;
  });
}

// LOAD TEXT
function loadText(index) {
  let textObj = texts[index];
  total = 0;
  completed = 0;
  answers = {};

  let parts = textObj.content.split(/\[(.*?)\]/);
  let display = "";

  for (let i = 0; i < parts.length; i++) {
    if (i % 2 === 1) {
      let id = "p" + total;
      display += `<span class="highlight" onclick="selectPart('${id}')">${parts[i]}</span>`;
      answers[id] = "";
      total++;
    } else {
      display += parts[i];
    }
  }

  total += 10; // all structure elements

  document.getElementById("textDisplay").innerHTML = display;
  document.getElementById("textTitle").innerText = textObj.title;

  document.getElementById("library").classList.add("hidden");
  document.getElementById("game").classList.remove("hidden");
}

// SELECT PART
function selectPart(id) {
  current = id;
  document.getElementById("answerBox").value = answers[id];
}

// WRITE ANSWER
document.getElementById("answerBox").addEventListener("input", function() {
  if (current) {
    if (answers[current] === "" && this.value !== "") completed++;
    answers[current] = this.value;
    updateProgress();
  }
});

// UPDATE PROGRESS
function updateProgress() {
  let extra = 0;
  document.querySelectorAll("textarea").forEach(t => {
    if (t.id !== "answerBox" && t.value !== "") extra++;
  });

  let percent = Math.round(((completed + extra) / total) * 100);
  document.getElementById("progress").innerText = "Progress: " + percent + "%";
}

// CHECK MISSING
function checkMissing() {
  let missing = [];

  for (let key in answers) {
    if (answers[key] === "") missing.push("analysis part");
  }

  document.querySelectorAll("textarea").forEach(t => {
    if (t.value === "") missing.push(t.id);
  });

  if (missing.length === 0) {
    alert("✅ Complete!");
  } else {
    alert("❌ Missing: " + missing.slice(0,5).join(", ") + "...");
  }
}

// BACK
function goBack() {
  document.getElementById("game").classList.add("hidden");
  document.getElementById("library").classList.remove("hidden");
}
